# Why: OMP does not auto-load Helix's managed status extension; wrap only
# interactive launch invocations so subcommands such as `omp config` keep
# their normal argv shape.
__helix_omp_should_skip_extension() {
  case "${1:-}" in
    'help'|'--help'|'-h'|'--version'|'-v') return 0 ;;
    '__complete'|'acp'|'agents'|'auth-broker'|'auth-gateway'|'bench'|'commit'|'completions'|'config'|'dry-balance'|'gallery'|'grep'|'grievances'|'install'|'join'|'models'|'plugin'|'read'|'say'|'search'|'setup'|'shell'|'ssh'|'stats'|'tiny-models'|'token'|'ttsr'|'update'|'usage'|'worktree'|'q'|'wt') return 0 ;;
  esac
  return 1
}
__helix_omp_cwd_is_usable() {
  local __helix_physical_cwd
  [[ -x . ]] || return 1
  if [[ -n "${PWD:-}" && -d "${PWD:-}" ]]; then
    [[ "${PWD}" -ef . ]]
  else
    # Why compare the path: shell builtins can print a cached path for a deleted cwd.
    __helix_physical_cwd="$(builtin pwd -P 2>/dev/null)" || return 1
    [[ -d "$__helix_physical_cwd" && "$__helix_physical_cwd" -ef . ]]
  fi
}
__helix_omp_invoke() {
  local __helix_use_extension="$1"
  shift
  if [[ $__helix_use_extension -eq 1 && -n "${HELIX_OMP_STATUS_EXTENSION:-}" && -f "${HELIX_OMP_STATUS_EXTENSION}" ]]; then
    if [[ "${1:-}" == "launch" ]]; then
      shift
      command omp launch --extension "${HELIX_OMP_STATUS_EXTENSION}" "$@"
    else
      command omp --extension "${HELIX_OMP_STATUS_EXTENSION}" "$@"
    fi
  else
    command omp "$@"
  fi
}
__helix_omp() {
  local __helix_use_extension=1
  __helix_omp_should_skip_extension "${1:-}" && __helix_use_extension=0
  if ! __helix_omp_cwd_is_usable; then
    local __helix_logical_cwd="${PWD:-${HELIX_WORKTREE_PATH:-${HELIX_ROOT_PATH:-}}}"
    # Why: a restored shell can retain the deleted directory inode after its path is recreated.
    (
      if [[ -z "$__helix_logical_cwd" ]]; then
        printf 'Helix: OMP cannot start because no terminal working directory is available. Open a new terminal in an existing directory.\n' >&2
        return 1
      fi
      if ! builtin cd -P -- "$__helix_logical_cwd" 2>/dev/null; then
        printf 'Helix: OMP cannot access the terminal working directory "%s". Open a new terminal in an existing directory.\n' "$__helix_logical_cwd" >&2
        return 1
      fi
      __helix_omp_invoke "$__helix_use_extension" "$@"
    )
  else
    __helix_omp_invoke "$__helix_use_extension" "$@"
  fi
}

  # an `alias omp` otherwise rewrites at parse time, aborting the rest of the file.
  function omp { __helix_omp "$@"; }

# Why: a typed alias expands inside the shell, after pane launch prep.
# Why unalias inside the substitution: an alias named codex makes command -v
# report the alias text, and the subshell leaves the user's own alias intact.
# Why || : twice — zsh alone aborts inside the substitution, but every shell's
# assignment adopts its exit status, so an absent codex trips set -e in bash too.
__helix_codex_binary="$(unalias codex 2>/dev/null || :; command -v codex 2>/dev/null || :)"
if [[ -n "${__helix_codex_binary:-}" && -x "${__helix_codex_binary}" ]]; then
  # Why the function reserved word: it suppresses alias expansion of the name,
  # which otherwise rewrites this header at parse time and aborts the whole file.
  function codex {
    # Why local: zsh's warn_create_global warns for each global a function creates.
    local __helix_codex_arg __helix_codex_isolate="${HELIX_CODEX_ISOLATE:-1}"
    if [[ -n "${HELIX_CODEX_LAUNCH_PREFLIGHT:-}" && -x "${HELIX_CODEX_LAUNCH_PREFLIGHT}" ]]; then
      "${HELIX_CODEX_LAUNCH_PREFLIGHT}" agent hooks prepare-codex >/dev/null 2>&1 || :
    fi
    for __helix_codex_arg in "$@"; do
      case "$__helix_codex_arg" in agents|queue|--no-daemon|--remote|--remote=*) __helix_codex_isolate=0 ;; esac
    done
    # Why probe every launch: a cached answer goes stale across an upgrade, and 0.155 and older exit 2 on the flag.
    if [[ "$__helix_codex_isolate" != 0 ]]; then
      case "$(command codex --help 2>/dev/null </dev/null)" in *--no-daemon*) set -- --no-daemon "$@" ;; esac
    fi
    command codex "$@"
  }
fi

# Why: SSH bash sessions need the same command lifecycle markers as local
# bash so agent rows stop showing "working" when the foreground command exits.
__helix_initializing_wrapper=1
__helix_osc133_precmd() {
  local exit_code=$?
  __helix_in_prompt_command=1
  if [[ -n "${__helix_in_command:-}" ]]; then
    printf "\033]133;D;%s\007" "$exit_code"
    unset __helix_in_command
  fi
  printf "\033]133;A\007"
  return "$exit_code"
}
__helix_osc133_prompt_done() {
  unset __helix_in_prompt_command; __helix_adopt_outer_debug_trap
  trap '__helix_osc133_preexec' DEBUG
}
__helix_osc133_preexec() {
  if [[ -n "${__helix_prompt_status_capture_command:-}" && "$BASH_COMMAND" == "$__helix_prompt_status_capture_command" ]]; then
    unset __helix_initial_prompt
    __helix_in_legacy_prompt_wrapper=1
    return 0
  fi
  if [[ -n "${__helix_initializing_wrapper:-}${__helix_in_debug_capture:-}${__helix_initial_prompt:-}${__helix_in_prompt_dispatch:-}${__helix_in_legacy_prompt_wrapper:-}${__helix_in_prompt_command:-}" ]]; then
    [[ -z "${__helix_initializing_wrapper:-}${__helix_in_debug_capture:-}" ]] || return 0
    if [[ -n "${__helix_initial_prompt:-}" && "$BASH_COMMAND" == "__helix_osc133_precmd" ]]; then
      unset __helix_initial_prompt; return 0
    fi
    if [[ -n "${__helix_in_prompt_dispatch:-}" ]]; then
      [[ -n "${__helix_dispatching_user_prompt_command:-}" ]] || return 0
      if [[ "${FUNCNAME[1]:-}" == "__helix_run_prompt_command_array" ]]; then
        case "$BASH_COMMAND" in
          '(( __helix_exit_code == 0 ))'|'__helix_restore_prompt_status "$__helix_exit_code"'|'eval "$__helix_prompt_part"'|'eval "$__helix_final_prompt_command"'|__helix_dispatching_user_prompt_command=*|__helix_osc133_precmd|__helix_osc133_prompt_done|__helix_prompt_mark) return 0 ;;
        esac
      fi
    elif [[ "${FUNCNAME[1]:-}" == "__helix_run_prompt_command_array" || "$BASH_COMMAND" == "__helix_run_prompt_command_array" ]]; then
      return 0
    fi
    [[ -z "${__helix_in_legacy_prompt_wrapper:-}" || -n "${__helix_dispatching_user_prompt_command:-}" ]] || return 0
    if [[ -n "${__helix_in_prompt_command:-}" && "$BASH_COMMAND" == "__helix_in_debug_capture=1" ]]; then
      return 0
    fi
  fi
  case "${FUNCNAME[1]:-}" in __helix_osc133_*|__helix_prompt_mark|__helix_restore_prompt_status) return 0 ;; esac
  case "$BASH_COMMAND" in __helix_osc133_precmd|__helix_osc133_prompt_done|__helix_prompt_mark) return 0 ;; esac
  __helix_run_user_debug_trap
  [[ -z "${__helix_in_prompt_command:-}" ]] || return 0
  [[ -z "${__helix_in_command:-}" ]] || return 0
  printf "\033]133;C\007"
  __helix_in_command=1
}
__helix_normalize_prompt_command_part() {
  local __helix_value="$1" __helix_output_name="$2" __helix_character __helix_chunk
  local __helix_value_length=${#1} __helix_suffix_length=0 __helix_backslash_length=0
  local __helix_output_length __helix_scan_start
  while (( __helix_value_length - __helix_suffix_length >= 1024 )); do
    __helix_scan_start=$(( __helix_value_length - __helix_suffix_length - 1024 ))
    __helix_chunk="${__helix_value:__helix_scan_start:1024}"
    case "$__helix_chunk" in
      *[!$' \t\n;']*) break ;;
      *) __helix_suffix_length=$(( __helix_suffix_length + 1024 )) ;;
    esac
  done
  while (( __helix_suffix_length < __helix_value_length )); do
    __helix_character="${__helix_value: -__helix_suffix_length - 1:1}"
    case "$__helix_character" in
      ' '|$'\t'|$'\n'|';') __helix_suffix_length=$(( __helix_suffix_length + 1 )) ;;
      *) break ;;
    esac
  done
  __helix_output_length=$(( ${#__helix_value} - __helix_suffix_length ))
  while (( __helix_output_length - __helix_backslash_length >= 1024 )); do
    __helix_scan_start=$(( __helix_output_length - __helix_backslash_length - 1024 ))
    __helix_chunk="${__helix_value:__helix_scan_start:1024}"
    case "$__helix_chunk" in
      *[!\\]*) break ;;
      *) __helix_backslash_length=$(( __helix_backslash_length + 1024 )) ;;
    esac
  done
  while (( __helix_backslash_length < __helix_output_length )); do
    __helix_character="${__helix_value:__helix_output_length - __helix_backslash_length - 1:1}"
    [[ "$__helix_character" == '\' ]] || break
    __helix_backslash_length=$(( __helix_backslash_length + 1 ))
  done
  # Preserve the first separator when an odd backslash run escapes it.
  if (( __helix_suffix_length > 0 && __helix_backslash_length % 2 == 1 )); then
    __helix_suffix_length=$(( __helix_suffix_length - 1 ))
    __helix_backslash_length=0
  fi
  __helix_output_length=$(( ${#__helix_value} - __helix_suffix_length ))
  __helix_value="${__helix_value:0:__helix_output_length}"
  # Bash 4.4-5.0 scalar prompt evaluation preserves an odd terminal backslash.
  if (( __helix_suffix_length == 0 && ((BASH_VERSINFO[0] == 4 && BASH_VERSINFO[1] >= 4) || (BASH_VERSINFO[0] == 5 && BASH_VERSINFO[1] == 0)) && __helix_backslash_length % 2 == 1 )); then
    __helix_value="$__helix_value\\"
  fi
  printf -v "$__helix_output_name" '%s' "$__helix_value"
}
__helix_restore_prompt_status() {
  return "$1"
}
__helix_update_user_debug_trap() {
  local __helix_debug_trap_spec="$1" __helix_unchanged_debug_trap_spec="$2"
  local __helix_debug_trap_command
  [[ "$__helix_debug_trap_spec" != "$__helix_unchanged_debug_trap_spec" ]] || return 0
  [[ "$__helix_debug_trap_spec" != "trap -- '__helix_osc133_preexec' DEBUG" ]] || return 0
  if [[ -z "$__helix_debug_trap_spec" ]]; then
    __helix_user_debug_trap=""
    unset __helix_chained_debug_trap
    return 0
  fi
  __helix_debug_trap_command="${__helix_debug_trap_spec#trap -- }"
  __helix_debug_trap_command="${__helix_debug_trap_command% DEBUG}"
  eval "__helix_user_debug_trap=$__helix_debug_trap_command"
  unset __helix_chained_debug_trap
}
__helix_run_user_debug_trap() {
  if [[ -n "${__helix_user_debug_trap:-}" ]]; then
    eval "$__helix_user_debug_trap" || true
  fi
}
__helix_adopt_outer_debug_trap() {
  local __helix_debug_trap_spec="${__helix_outer_debug_trap_spec:-}"
  unset __helix_outer_debug_trap_spec
  __helix_update_user_debug_trap "$__helix_debug_trap_spec" "trap -- '__helix_osc133_preexec' DEBUG"
}
__helix_run_prompt_command_array() {
  local __helix_exit_code="${__helix_prompt_status:-$?}" __helix_prompt_part __helix_prompt_index __helix_user_count
  local __helix_suffix_part
  local __helix_final_prompt_command
  local __helix_in_prompt_dispatch=1 __helix_dispatching_user_prompt_command=""
  unset __helix_prompt_status
  __helix_adopt_outer_debug_trap
  trap '__helix_osc133_preexec' DEBUG
  for __helix_prompt_part in "${__helix_prompt_command_prefix[@]+"${__helix_prompt_command_prefix[@]}"}"; do
    if (( __helix_exit_code == 0 )); then
      eval "$__helix_prompt_part"
    else
      __helix_restore_prompt_status "$__helix_exit_code" || eval "$__helix_prompt_part"
    fi
  done
  __helix_user_count=0
  for __helix_prompt_part in "${__helix_prompt_command_array[@]+"${__helix_prompt_command_array[@]}"}"; do
    __helix_user_count=$(( __helix_user_count + 1 ))
  done
  for (( __helix_prompt_index = 0; __helix_prompt_index + 1 < __helix_user_count; __helix_prompt_index++ )); do
    __helix_prompt_part="${__helix_prompt_command_array[__helix_prompt_index]}"
    __helix_dispatching_user_prompt_command=1
    if (( __helix_exit_code == 0 )); then
      eval "$__helix_prompt_part"
    else
      __helix_restore_prompt_status "$__helix_exit_code" || eval "$__helix_prompt_part"
    fi
    __helix_dispatching_user_prompt_command=""
  done
  if (( __helix_user_count > 0 )); then
    __helix_prompt_part="${__helix_prompt_command_array[__helix_user_count - 1]}"
    # Why: keep the final user hook and Helix suffixes in one status-preserving eval.
    __helix_final_prompt_command='eval "$__helix_prompt_part"'
    for __helix_suffix_part in "${__helix_prompt_command_suffix[@]+"${__helix_prompt_command_suffix[@]}"}"; do
      __helix_final_prompt_command+=$'\n'"$__helix_suffix_part"
    done
    __helix_dispatching_user_prompt_command=1
    if (( __helix_exit_code == 0 )); then
      eval "$__helix_final_prompt_command"
    else
      __helix_restore_prompt_status "$__helix_exit_code" || eval "$__helix_final_prompt_command"
    fi
    __helix_dispatching_user_prompt_command=""
  else
    for __helix_prompt_part in "${__helix_prompt_command_suffix[@]+"${__helix_prompt_command_suffix[@]}"}"; do
      if (( __helix_exit_code == 0 )); then
        eval "$__helix_prompt_part"
      else
        __helix_restore_prompt_status "$__helix_exit_code" || eval "$__helix_prompt_part"
      fi
    done
  fi
  return "$__helix_exit_code"
}
__helix_finish_legacy_prompt_dispatch() {
  local __helix_suffix_part
  if [[ -n "${__helix_in_prompt_command:-}" ]]; then
    for __helix_suffix_part in "${__helix_prompt_command_suffix[@]+"${__helix_prompt_command_suffix[@]}"}"; do
      eval "$__helix_suffix_part"
    done
  fi
  trap '__helix_osc133_preexec' DEBUG
  unset __helix_in_legacy_prompt_wrapper
}
__helix_normalize_prompt_command() {
  [[ -z "${__helix_prompt_command_normalized:-}" ]] || return 0
  local __helix_prompt_part
  local -a __helix_normalized=()
  for __helix_prompt_part in "${PROMPT_COMMAND[@]+"${PROMPT_COMMAND[@]}"}"; do
    __helix_normalize_prompt_command_part "$__helix_prompt_part" __helix_prompt_part
    [[ -n "$__helix_prompt_part" ]] && __helix_normalized+=("$__helix_prompt_part")
  done
  __helix_prompt_command_normalized=1
  if (( BASH_VERSINFO[0] > 5 || (BASH_VERSINFO[0] == 5 && BASH_VERSINFO[1] >= 1) )); then
    PROMPT_COMMAND=("${__helix_normalized[@]+"${__helix_normalized[@]}"}")
  else
    __helix_prompt_command_array=("${__helix_normalized[@]+"${__helix_normalized[@]}"}")
    __helix_prompt_command_prefix=()
    __helix_prompt_command_suffix=()
    unset PROMPT_COMMAND
    # Why: PID scope distinguishes legacy prompt dispatch from ordinary user command text.
    __helix_prompt_status_variable="__helix_prompt_status_$$"
    __helix_prompt_status_capture_command="$__helix_prompt_status_variable=\$?"
    __helix_prompt_status_value="\${$__helix_prompt_status_variable}"
    PROMPT_COMMAND="$__helix_prompt_status_capture_command; __helix_prompt_status=$__helix_prompt_status_value"'; __helix_prompt_had_functrace=""; if [[ -o functrace ]]; then __helix_prompt_had_functrace=1; set +T; fi; __helix_outer_debug_trap_spec="$(trap -p DEBUG)"; [[ -z "$__helix_prompt_had_functrace" ]] || set -T; unset __helix_prompt_had_functrace; __helix_run_prompt_command_array; __helix_finish_legacy_prompt_dispatch'
  fi
}
__helix_prepend_prompt_command() {
  local command="$1"
  __helix_normalize_prompt_command
  if (( BASH_VERSINFO[0] > 5 || (BASH_VERSINFO[0] == 5 && BASH_VERSINFO[1] >= 1) )); then
    PROMPT_COMMAND=("$command" "${PROMPT_COMMAND[@]+"${PROMPT_COMMAND[@]}"}")
  else
    __helix_prompt_command_prefix=("$command" "${__helix_prompt_command_prefix[@]+"${__helix_prompt_command_prefix[@]}"}")
  fi
}
__helix_append_prompt_command() {
  local command="$1"
  __helix_normalize_prompt_command
  if (( BASH_VERSINFO[0] > 5 || (BASH_VERSINFO[0] == 5 && BASH_VERSINFO[1] >= 1) )); then
    PROMPT_COMMAND+=("$command")
  else
    __helix_prompt_command_suffix+=("$command")
  fi
}
